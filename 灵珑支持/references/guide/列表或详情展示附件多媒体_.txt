三、详情/列表  状态控制
在组件渲染前执行，将后端返回的相对路径转换为完整可访问 URL，格式化数据结构适配平台多媒体组件，并在无媒体时自动隐藏组件。
适用场景：需要根据数据动态显示 / 隐藏组件
1.首先开启图标显示
2.状态控制编写脚本  (一定要打开 控制开关)
function main(params) 
    // params 为当前卡片数据
    // NORMAL 普通状态
    // HIDDEN 隐藏状态
    console.log("params =>", params);

    
    if (params && params["多媒体"] && params["多媒体"].value) {
        const origin = window.location.origin;
        const apiPrefix = '/linglongZT-api/unity/establish/file';
        const baseUrl = origin + apiPrefix;

        const appId = '2061264891226980352';
        const accessToken = api.APP_TOKEN();

        // 直接取单个文件对象
        const fileObj = params["多媒体"].value;

        const fileId = fileObj.fileId;
        const serviceType = fileObj.serviceType;


        // 拼接预览地址
        const previewUrl = `${baseUrl}/${fileId}/preview?appId=${appId}&serviceType=${serviceType}&access_token=${accessToken}`;
        console.log("previewUrl", previewUrl)
        params["多媒体"].value = [
            {
                url: previewUrl,
                mediaType: '1',     
            }
        ];
    }

    // 无数据隐藏
    if (
        params &&
        params["多媒体"] &&
        (!params["多媒体"].value || params["多媒体"].value.length === 0)
    ) {
        return "HIDDEN";
    }
    return "NORMAL";
}

四、常见问题与排查指南
4.1 图片无法加载
打开浏览器控制台（F12），查看Network面板中图片请求的 URL 是否正确
检查接口前缀是否与当前环境匹配
确认后端返回的media_url是否为相对路径（不含域名和前缀）
排查是否存在跨域问题
4.2 组件不显示 / 不隐藏
检查组件绑定的字段名是否与代码中一致
确认后端返回的数据结构是否符合要求
查看控制台是否有 JS 报错，重点检查数组遍历相关的错误
4.3 多文件渲染异常
确认多文件分隔符为英文逗号 ,，而非中文逗号
检查每个文件的media_url是否都不为空
若使用公式生成路径，确认REPLACE_TO函数的替换逻辑正确
