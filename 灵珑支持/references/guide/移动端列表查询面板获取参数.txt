function main() {
  const params = {
    startTime: '2024-09-01 00:00:00',
    endTime: '2024-09-30 23:59:59',
    regionCode: '11',
    regionArr: ['', '1', '11'] // 级联选择情况
  }; // 跳转参数
  this.$router.push({
        name: "page-preview",// 页面
        query: {
            title: "案件列表",
            pageId: '1774679672873394176', // 对应页面id
            showHeader: this.$route.query.showHeader,
            isFromAction: 1,
            _t: new Date().getTime() + "",
        },
        params: params
    });
}

if (param.condition.tableConditionDescriptor) {function main(pageId, param) {

            param.condition.params = {
              startTime: '',
              endTime: '',
              regionCode: '',
              ...默认接口参数
            };
            if (params) {
                Object.assign(param.condition.params, params) // 把跳转的参数和接口参数进行拼接汇总
            }

            /** 若查询面板存在筛选条件, 函数的param.condition参数中会存在tableConditionDescriptor 这里判断tableConditionDescriptor是否存在，
            若存在接口参数就在这个判断里面更新, 若查询字段未选择或未输入任何值, tableConditionDescriptor.members中就没有这个查询字段对象
            每个查询字段都有自己的field标识, 需要自己打印出来查看 **/
            if (param.condition.tableConditionDescriptor) {
                let queryParams = param.condition.tableConditionDescriptor.members;
                queryParams.map(item => {
                    if (item.field === "542648d4-54dc-4eb5-ba2d-c99c1074c747") {
                        param.condition.params.startTime = item.values[0] ? item.values[0] + ' 00:00:00' : params.startTime;
                        param.condition.params.endTime = item.values[1] ? item.values[1] + ' 23:59:59' : params.endTime;
                    }
                    if (item.field === "d872a1f2-0898-4213-89a0-9ff3f808956a") {
                        param.condition.params.regionCode = item.values[item.values.length - 1];
                    } else {
                        param.condition.params.regionCode = '';
                    }
                })
            }

            api.API_DATA_EXECUTE("7adcc63a-137d-437f-81c4-af56fba1afca", "BUNCHES", param, { lostAbility: "00" })
                .then((res) => {

                    resolve(newRes);
                }).catch(() => {

解决方法：拿到跳转传递的参数，例如时间参数，然后给查询字段赋值。
