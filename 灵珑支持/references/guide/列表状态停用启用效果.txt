列表状态停用启用效果
1 使用场景
在列表页面的操作栏中会出现根据列表中的某一字段来控制操作列中按钮的显示对应的操作内容，如在列表信息为停用时显示启用。
2 效果
当列表信息为停用时，显示启用按钮，点击启用按钮可以使列表状态为启用状态；当列表信息为启用时，显示停用按钮，点击停用按钮可以使列表状态为停用状态。

// 通过返回值来判断：如果true 则是JSON格式，否则，不是JSON格式
function IsJsonString(str) {
    try {
        JSON.parse(str);
    } catch (e) {
        return false;
    }
    return true
}
2、操作列中同时设置停用和启用的按钮，用JS脚本控制按钮的显隐
function main(context) {
    // NORMAL 普通状态
    // HIDDEN 隐藏状态
    // DISABLED 禁用状态
    let currentDatas = context.currentDatas || [];
    // 拿到当前行数据
    let row = currentDatas[0];
    // 找到当前行中字段名为state的数据
    let idData = row.find(r => r.name === 'enable_flag');
    //判断state字段为0则显示按钮，否则不显示按钮
    if (idData.value == '启用') {
        return 'NORMAL';
    } else {

        return "HIDDEN";
    }
}
