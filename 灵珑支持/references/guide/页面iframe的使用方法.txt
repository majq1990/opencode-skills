页面iframe的使用方法
1 使用场景
在进行配置页面的时候会存在复用其他系统的页面，因此使用iframe来进行嵌套使用其他页面。
2 效果
在页面中嵌套使用其他已经配置好或存在的页面。
3 配置
新建页面，在布局组件中拖入iframe部件
点击页面，点开右下角绑定动作，按图所示绑定页面动作

export function didMount() {
  debugger
  console.log("this.state.extraParams");
  console.log(this.state.extraParams);
  let url = "http://10.255.18.71:58080/dist/index.html#/user-detail";//需要展示的页面连接
  if (this.state.extraParams) {
    var extraParams = this.state.extraParams;
    url += "?detailFrom=2";
    url += "&fromID=" + extraParams.dutyDeptId;
    url += "&fromName=" + extraParams.dutyDeptName;//传递的参数名称需要跟对应的列表传参的参数一一对应
    url += "&v22_token=" + extraParams.v22_token;
    url += "&startTime=" + encodeURI(extraParams.startTime) + "&endTime=" + encodeURI(extraParams.endTime);
  }

  var interfaceUrl = "http://10.255.18.71:58080/eUrbanMIS/perceptionserver"

  url += "&interfaceUrl=" + interfaceUrl;

  this.$$model("basic_iframe_9199uh").props.url = url;//当前页面的iframe组件标识
  console.log(url);
  console.log("「页面 JS」：当前页面地址 " + location.href);
}
