SELECT
    node_id AS id,
    senior_id,
    node_name AS name,
    node_type
FROM
    (
        SELECT
            CONCAT ('R_', id) AS node_id,
            CONCAT ('R_', senior_id) AS senior_id,
            name AS node_name,
            1 AS node_type
        FROM
            sys_region
        where type < 3
        UNION ALL
        SELECT
            CONCAT ('U_', unit_id) AS node_id,
            CASE
                WHEN senior_id = 0 OR senior_id IS NULL THEN CONCAT ('R_', region_id)
                ELSE CONCAT ('U_', senior_id)
            END AS senior_id,
            unit_name AS node_name,
            2 AS node_type
        FROM
            law_unit_extend_info
        WHERE
            delete_flag = 0
    ) AS tree
ORDER BY
    node_type,
    id

function main(defaultParams) {
    const self = this;
    const formData = getFormData(self);

    getLawStallAppointmentRecordInfo(formData)
        .then(function (appointmentRecords) {
            // 返回一个 Promise 链，而不是直接抛出错误
            return checkAppointmentConflict.call(self, formData, appointmentRecords)
                .then(function (hasConflict) {
                    if (hasConflict) {
                        // 存在冲突，返回一个特殊标记，后续处理会跳过保存步骤
                        return {type: 'conflict', conflict: true};
                    }
                    // 没有冲突，继续保存表单
                    return saveFormData.call(self, formData)
                        .then(function (saveResult) {
                            return {type: 'save', result: saveResult};
                        });
                });
        })
        .then(function (operationResult) {
            // 根据上一步的结果决定后续操作
            if (operationResult.type === 'conflict') {
                // 存在冲突，直接返回，不继续执行
                return Promise.reject({name: 'AppointmentConflictError', message: '存在预约冲突'});
            }

            if (operationResult.type === 'save' && operationResult.result) {
                // 保存成功，添加详细记录
                return addDetailedAppointmentRecords.call(operationResult.result.result.documentNo);
            }

            // 保存失败
            return Promise.reject(new Error('表单保存失败'));
        })
        .then(function () {
            self.$message.success('预约成功');
            openPage(self, '1955802345355870208', '我的预约', formData.holderIdValue, '')
            
        })
        .catch(function (error) {
            // 分离错误处理：冲突错误已经提示过，不再显示通用错误
            if (error.name !== 'AppointmentConflictError') {
                console.error('摊位预约处理失败:', error);
                self.$message.error('预约处理失败，请稍后重试');
            }
        });
}

 * 检查预约冲突
 * @returns {Promise<boolean>} true-存在冲突，false-无冲突
 */
function checkAppointmentConflict(formData, appointmentRecords) {
    const self = this;

    return new Promise(function (resolve) {
        if (!appointmentRecords || appointmentRecords.length === 0) {
            resolve(false);
            return;
        }

        // 查找与当前预约时段冲突的记录
        const conflictRecord = findConflictRecord(formData, appointmentRecords);

        if (!conflictRecord) {
            resolve(false);
            return;
        }

        const appointmentType = conflictRecord.appointment_type;
        const appointmentDate = conflictRecord.appointment_date;
        const stallCode = conflictRecord.stall_code;

        if (appointmentType == 1) {
            self.$message.warning("当前摊位 " + appointmentDate + " 已被预约");
        } else {
            self.$message.warning("您已预约 " + appointmentDate + " 的摊位 " + stallCode);
        }

        resolve(true);
    });
}

/**
 * 查找冲突记录 - 改进版本
 */
function findConflictRecord(formData, appointmentRecords) {
    const appointmentRecordInfo = appointmentRecords.find(item => formData.stallUsePeriodType.includes(item.stall_use_period_type));
    return appointmentRecordInfo;
}
