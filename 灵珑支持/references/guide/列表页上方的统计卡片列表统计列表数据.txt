列表页上方的统计卡片列表统计列表数据
使用场景
在一个页面中存在需要根据数据的某一类型对数据使用卡片的形式进行数据展示，同时在点击不同类型对应的卡片时可以在列表中显示对应的数据。
效果
用户点击卡片，在下方的数据列表中能够根据卡片所对应的类型显示对应的数据
配置
基于列表的实体表新建表单用来做卡片列表的数据展示
选择组件组件为市政统计卡片，数据源根据实际业务需求可选择实体数据、网关数据，本案例采用网关数据获取数据。
配置数据源添加数据接口（api已在API管理中配置）
Query参数为请求的参数，本示例以街道过滤为示例
接口函数说明：

function fit(result) {
  // fit用于成功解析数据源响应后回调
  // result为数据源返回结果
  // 一般为JSON字符串，建议先进行JSON.parse(result)操作)
  // 若要配置到select的option上则需要保证返回结构包含label及value属性
  console.log("fit",result);
  var dataListJson = JSON.parse(result).result;
  const litiantList = ['全部', '法人和其他组织', '个体工商户', '公民']
  const litiantValueList = [dataListJson[0].count || 0, dataListJson[2].count || 0, dataListJson[3].count || 0, dataListJson[1].count || 0]
  const listValue = [];
  for (let i = 0; i < 4; i++) {
    listValue.push({
      label: litiantValueList[i],
      num: litiantList[i]
    })
  }
  return listValue;
}
市政统计卡片组件会读取fit() 函数返回的结果，因此在进行数据展示时候需要在fit返回数据前对数据进行预处理，使得返回的结果集为键值对的形式存在即为listValue，其中lable和num为组件的属性映射
function didFetch(result) {
  // didFetch用于请求完成后回调

  return result;
}

function main(item, index, dataList) {

    var self = this;
    let street_region_id = self.$$("text_input_8r67nd").$$getValue();
    var rootPageElement = self.$$roots()[0];

    let regions = street_region_id.split(",").map(Number)

    let LitigantCodeNum = "";

    let pageListAll = rootPageElement.$$("page_widget_list_1evnzt");
    let pageListLegal = rootPageElement.$$("page_widget_list_dtdsog");
    let pageListMannger = rootPageElement.$$("page_widget_list_f9jfcg");
    let pageListPerson = rootPageElement.$$("page_widget_list_62kq3f");
    let pageTableList = null;

    pageListAll.$el.style.display = "none";
    pageListMannger.$el.style.display = "none";
    pageListPerson.$el.style.display = "none";
    pageListLegal.$el.style.display = "none";
    if (index === 0) {
        // 全部

        pageListAll.$el.style.display = "block";
        // pageListLegal.$el.style.display = "none";
        pageTableList = pageListAll;

    } else if (index === 1) {
        // 法人和其他组织
        LitigantCodeNum = 2;
        // pageListAll.$el.style.display = "none";
        // pageListMannger.$el.style.display = "none";
        // pageListPerson.$el.style.display = "none";
        pageListLegal.$el.style.display = "block";
        pageTableList = pageListLegal;

    } else if (index === 2) {
        // 个体工商户
        LitigantCodeNum = 3;
        // pageListAll.$el.style.display = "none";
        pageListMannger.$el.style.display = "block";
        // pageListPerson.$el.style.display = "none";
        // pageListLegal.$el.style.display = "none";
        pageTableList = pageListMannger;

    } else if (index === 3) {
        // 公民
        LitigantCodeNum = 1;
        // pageListAll.$el.style.display = "none";
        // pageListMannger.$el.style.display = "none";
        pageListPerson.$el.style.display = "block";
        // pageListLegal.$el.style.display = "none";
        pageTableList = pageListPerson;
    }

    if (pageTableList && pageTableList.dataEngine && pageTableList.dataEngine.state) {
        const tableComp = pageTableList.$$("table_bqb9q2");

        if (tableComp) {
            var members = pageTableList.dataEngine.state.layout.props.dataSet.filterData.filter.members.slice(0, 1);
            members.push({
                field: "950f03ae-b913-40d9-ae1e-617c5aea2e88",
                fieldName: "street_region_id",
                "fieldType": "TYPE_INTEGER",
                valueType: "Constant",
                values: regions || [],
                value: "",
                operate: "In", // 范围查询
                className: "Single",
                treeUnitIdFiter: "1"
            });
            var filter = {
                "combine": "And",
                "className": "List",
                "members": members
            };

            pageTableList.dataEngine.state.layout.props.dataSet.filterData = {
                "enabled": true,
                "filter": filter,
                "filterType": "FILTER",
                "rawCondition": {},
                "rawFormula": ""
            }





            selectByLitigantCode(pageTableList, LitigantCodeNum);
            window.bigLitigantCode = index


            tableComp.onRefresh();
        }
    }


    return true;
}

function selectByLitigantCode(pageTableList, litigant_big_type_code) {

    var members = pageTableList.dataEngine.state.layout.props.dataSet.filterData.filter.members.slice(0, 2);

    members.push({
        field: "2b2ff819-7dd4-4ae5-bd17-8ec95db3a6e9",
        fieldName: "litigant_big_type_code",
        "fieldType": "TYPE_INTEGER",
        valueType: "Constant",
        values: [],
        value: litigant_big_type_code || null,
        operate: "In", // 范围查询
        className: "Single"
    });



    var filter = {
        "combine": "And",
        "className": "List",
        "members": members
    };
    pageTableList.dataEngine.state.layout.props.dataSet.filterData.filter = filter;
}
