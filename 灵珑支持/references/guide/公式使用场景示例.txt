function main(defaultParams) {

    var self = this; // 重置作用域
    var tableComp = self.findElement("table_4keoyy"); // 获取表格元素
    tableComp.loading = true; // 使表格置于loading状态

    api.EXEC_FORMULA("INSERT('t_e7811c95737a4019a3d7c5315dd14b39',['id',UUID(),'name','" + defaultParams["f_cf3b1598edb64c6b9ef513a745f63bbd"] + "','address','" + defaultParams["f_e79ee6a7cfc74372a1f0c8c416c0c438"] + "'])").then((res) => {
        // then() 为调用成功的回调
        if (!res.hasError) {

            tableComp.$$refreshData().then(() => {
                tableComp.loading = false; // 将表格置为非loading状态
                self.$message.success("已收藏！");
            }); // 调用表格的数据刷新动作

        } else {
            self.$message.error(res); // 显示错误提示信息
            tableComp.loading = false; // 将表格置为非loading状态
        }
        
    }).catch(err => {
        self.$message.error(err.message); // 显示错误提示信息
        
    });


}

实现效果
列表部件的“条件过滤”引用REGION_CHILD_LIST公式返回区域表里多级结果记录。
配置说明
1、公式本身验证：
普通表单：“单行文本”默认值为“公式编辑”。
注意：如果只有2级下级，而level设置为大于最在级数时可以返回当前最大级数的全部内容。
全部主要的组合公式配置和结果：
2、列表内容通过公式过滤的配置
（1）统一用户中心-基础数据-行政区划：按级别配置行政区划，数据保存在表com_region表。如下图：
（2）人员配置行政区划

（3）应用设计-基础配置-会话变量配置：新增会议变量，
会话命名：请取有意义的名称
配置公式：
REGION_CHILD_LIST:获取区划com_region某父节点下的所有子节点的某字段列表(有缓存)，
    REGION_CHILD_LIST('regionId', level, includeSelf, 'filed')。
        regionId填字符串或填null时取当前登录人的区划；
        level填数字，表示取父节点包含几级的子节点；第二个参数得大于0或者为null
        includeSelf 填true或false，表示返回值是否包含该父节点；
        filed填字符串，表示返回某个字段的列表，如'id'、'name'、'code'、'grade'，填null时返回全部字段信息
（4）列表部件：“条件过滤”里配置过滤条件，选择“会话变量配置”里配置的公式
注意：在“会话变量配置”里通过【编辑】修改公式内容后，列表部件的“条件过滤”需要重新选择下图的公式，否则不生效。
如果修改后过滤结果仍不对，建议删除过滤条件，再重新配置。

6、返回登录人员的区划信息
实现效果
参考案件 #281724。
表单部件中文本组件的“公式编辑”引用REGION_BY_GRADE公式，返回登录人员的区划信息。
配置说明
1、统一用户中心配置所属区划
当前登录人所属区划取的是用户中心的数据，所以确保外部集成系统如mis的用户区划信息同步至用户中心；如果用户体系在灵珑用户中心维护，则需要维护区划信息及用户与区划的关联关系。
在用户中心->用户列表下编辑用户，设置所属区域。
2、表单中的组件配置公式
公式格式：REGION_BY_GRADE('regionId', level, 'mode', 'returnFiled')
示例：
通过com_region的grade字段级别值查询。
区划表com_region中grade值定义为：1省级、2市级、3区县级、4街道级别、5社区级、6网格、7其他。
查询省级区划：REGION_BY_GRADE(REGION_ID(), 1, 'grade', 'name')
在数据管理->新增数据时，可见默认填充的区划信息。
