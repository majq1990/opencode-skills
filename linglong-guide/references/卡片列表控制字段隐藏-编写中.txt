1、新建一个数据模型，主要用documentNo关联查询到对应的数据(灵珑媒体表)；
select
"ID" as id,
"DOCUMENTNO" as documentNo,
"PAGEID" as pageId,
"VERSION" as version,
"UNIQUEID" as uniqueId,
"UNIQUEINDEX" as uniqueIndex,
"UNIQUECODE" as uniqueCode,
"FILEID" as fileId,
"FILENAME" as fileName,
"FILESIZE" as fileSize,
"MEDIATYPE" as mediaType,
"CREATOR" as creator,
"CREATETIME" as createTime,
"DATAID" as dateId,
"SERVICETYPE" as serviceType
from lcpage_form_datum
where documentNo = ${id}
