export function didMount() {
  console.log("「页面 JS」：当前页面地址 " + location.href);
  //接收页面传过来的轨迹字段sim_card_num
  let params = this.$route;
  let vehicleId = params.query.sim_card_num;

  //设置时间为当前日期
  console.log(params);
  let date = new Date();
  let year = date.getFullYear();
  let month = date.getMonth() + 1;
  let day = date.getDate();
  let currentDate = `${year}-${month}-${day}`;

  this.$$("map_base_6immbz").$$showLocalObjectTrace("traceLayer_45cme9", vehicleId,
    `${currentDate} 07:00`, `${currentDate} 23:00`)
}
