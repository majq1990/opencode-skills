页面中实现按照单独用户实现收藏功能和取消收藏功能
1 使用场景
在用户管理中会出现需要出现用户对某条数据进行收藏的情况，按照需求需要按照用户对收藏进行隔离操作，A用户看不到B用户的收藏，A用户的收藏和B用户的收藏为独立的收藏内容，互相不干扰。
抽象出来业务逻辑为，当用户点击收藏后在收藏表里插入一对应的收藏字段，下面例子插入收藏人和 举报人的字段。取消收藏按照业务需求来只需要删除掉对应的字段。同时需要在视图里面利用 case when 增加一个类型字段用来判定是否当前用户已经收藏，最后在列表中配置相应的过滤条件。
2 效果
用户点击收藏按钮，列表刷新，收藏按钮隐藏，显示取消收藏按钮，同时我的收藏里面显示当前收藏的字段，在其他的列表点击取消收藏后，实时更新为收藏字段同时我的收藏列表隐藏掉当前数据
3 配置
在视图中通过左连接拼接收藏表，并且增加对应的变量字段，按照业务需求配置IF判断语句；参考示例代码如下

    mis_reporter_remark mrr ON mrr.reporter_id = mr.reporter_id

function main(params) {
    var self = this; // 重置作用域
    var tableComp = self.$$element("table_4okw6w"); // 获取表格元素
    //找到对应的列表reporter_id
    var reporterId = params['d09418b9-277c-49b8-a090-4543f2d997d4'];
    var reporterName = params['e35e6af0-ee31-4836-bc04-32b36afb659a']
    //定义收藏时间
    var currentDate = new Date();
    const create_time = currentDate.getFullYear() + '-' + (currentDate.getMonth() + 1) + '-' + currentDate.getDate() + ' ' + currentDate.getHours() + ':' + currentDate.getMinutes() + ':' + currentDate.getSeconds()
    //操作人
    var userInfo = JSON.parse(window.localStorage.getItem("linglong:user"));
    let userName = userInfo.user.username
    console.log('userInfo', userInfo)
    api.EXEC_FORMULA("UPSERT('t_580ba97c3337411d8ad3782e0ba71951',QUERY_AND(QUERY_EQ('reporter_id', '" + reporterId + "'),QUERY_EQ('human_name', '" + userName + "')),'', ['reporter_name','" + reporterName + "','reporter_id', '" + reporterId + "', 'human_name', '" + userName + "', 'create_time', '" + create_time + "'], ['reporter_id', 'human_name'])").then((res) => {
        if (!res.hasError) {
            tableComp.$$refreshData().then(() => {
                console.log("保存成功")
            });
        } else {
            self.$message.error(res); // 显示错误提示信息
        }
    }).catch(err => {
        self.$message.error(err.message); // 显示错误提示信息
    });
}
