全局方法注册于表单部件didMount事件：
function main() {
    // 卡片id
    const COMPONENT_IDS = {
        REC_TYPE: "com_board_table_23n9ug" // 案件类型
    }
    // 公共筛选组件id
    const COMMON_PARAMS_CMP_ID = {
        DATE_TAB: "csdn_form_custom_date_5jic6i", // 时间选择
        REGION_CASCADER: "csdn_form_field_cascader_3yt91k", // 区域级联选择
SINGLE_SELECT:"csdn_single_select_apkxm" // 下拉单选
    }
    window.COMPONENT_IDS = COMPONENT_IDS
    window.COMMON_PARAMS_CMP_ID = COMMON_PARAMS_CMP_ID
    let self = this;
    /**
     * 获取时间参数
     */
    function getTimePara() {
        const timeCmp = self.$$(COMMON_PARAMS_CMP_ID.DATE_TAB)
        const { currentTime, selectDate } = timeCmp || {};
        let startTime = "";
        let endTime = ""
        let tabs = []
        if (currentTime) {
            startTime = currentTime.start.format("YYYY-MM-DD 00:00:00")
            endTime = currentTime.end.format("YYYY-MM-DD 00:00:00")
            tabs = timeCmp.data.props.tabs
        }

        return {
            startTime,
            endTime,
            currentTime,
            selectDate,
            tabs
        }
    }
    /**
     * 获取区域参数
     */
    function getRegionParam() {
        const regionCmp = self.$$(COMMON_PARAMS_CMP_ID.REGION_CASCADER) || {}
        let regionCode = ""
        let regionId = ""
        let regionArr = []
        if (regionCmp) {
            const { itemOptions = {} } = regionCmp.textData.props.defaultValue || {};
            regionCode = itemOptions.regionCode;
            regionId = itemOptions.regionId;
            regionArr = regionCmp.defaultValue
        }
        return {
            regionCode,
            regionArr,
            regionId
        }
    }
    window.GLOBAL_PARAM_TIME = getTimePara();
    window.GLOBAL_PARAM_REGION = getRegionParam();

    /**
     * 全局获取参数
     */
    window.GET_GLOBAL_PARAM = (id) => {
        const { startTime, endTime } = getTimePara();
        const { regionCode, regionArr,regionId } = getRegionParam();
        // 公共筛选组件的参数
        let commonParams = {
            startTime,
            endTime,
            startCreateTime: startTime,
            endCreateTime: endTime,
            regionCode,
            regionArr,
            regionId
        }
        // 这里可以根据传入的组件id，写某个组件需要用到的特殊参数
        // if (id === COMPONENT_IDS.BUSINESS_TYPE) {
        //     let tabType = self.$$("csdn_form_tab_container_3u4tiu").data.props.value
        //     Object.assign(commonParams, {
        //         tab: tabType === 1 ? '二级分类' : '三级分类'
        //     })
        // }
        // console.log("获取参数", commonParams)

        return commonParams
    }
    /**
     * 根据id设置组件加载状态
     * id 组件id
     * loading 加载状态 true/false
     * list 加载完后的数据
     */
    window.SET_CMP_LOADING = (id, loading, list) => {
        const cmp = self.$$(id);
        if (cmp) {
            // 设置加载状态
            cmp.data.props.loadingConfig.loading = loading;
            // 组件赋值
            if (list && list.length) {
                cmp.data.props.dataSource.options = list || []

            }
        } else {
            console.warn("SET_CMP_LOADING----未找到组件实例:" + id)
        }
    }
    /**
     * 刷新所有组件
     */
    window.REFRESH_ALL = () => {
        for (const key in COMPONENT_IDS) {
            if (Object.prototype.hasOwnProperty.call(COMPONENT_IDS, key)) {
                const id = COMPONENT_IDS[key];
                refresh(id)
            }
        }
    }
    /**
     * 根据id刷新某个组件
     */
    function refresh(id) {
        // debugger
        const cmp = self.$$(id);
        if (cmp) {
           cmp.$$executeApiDataSource(cmp.data.props.dataSource.apiId)
        } else {
            console.warn("REFRESH_CMP---未找到组件实例:" + id )
        }
    }
    window.REFRESH_CMP = refresh;
}
常量声明
window.COMPONENT_IDS 用于声明组件id
window.COMMON_PARAMS_CMP_ID用于声明全局筛选组件
声明格式为 {大写key: "组件id"} // 备注组件业务名称
比如业务类型卡片里面的这个组件，声明为
{
        BUSINESS_TYPE: "csdn_form_chart_category_9qpabg" // 业务类型
}
其他地方需要用时，直接通过window.COMPONENT_IDS.BUSINESS_TYPE调用。不允许在脚本的其他地方出现组件标识。

function fit(result) {
  // fit用于成功解析数据源响应后回调
  // result为数据源返回结果
  // console.log(result,"result")
  // 如果是api接口 则需要进行数据格式的转换
  const newRes = JSON.parse(result).result;
  /* 获取区划 */
  let filterData = (data, needAll, parent) => {
    let list = (data || []).map(item => {
      return {
        value: item.regionCode,
        label: item.regionName,
        id: item.id,
        text: item.regionName,
        regionType: item.regionType,
        regionId: item.id,
        regionCode: item.regionCode,
        children: item.children && item.children.length ? filterData(item.children, true, item) : ""
      }
    });

    if (needAll) {
      list.unshift({
        value: parent.regionCode,
        label: "全部",
        id: parent.id,
        text: "全部",
        regionType: parent.type,
        regionId: parent.id,
        regionCode: parent.regionCode,
      })
    }

    return list;
  }
  // 一般为JSON字符串，建议先进行JSON.parse(result)操作)
  // 若要配置到select的option上则需要保证返回结构包含label及value属性
  const newList = filterData(newRes)
  return newList;
}

请求发送处理前函数
function willFetch(queryParams, formParams, json, requestHeaders) {
    // 可以更改queryParams, formParams, json, requestHeaders参数


    // 调用全局获取参数的方法
    if (window.GET_GLOBAL_PARAM) {
        const params = window.GET_GLOBAL_PARAM(window.COMPONENT_IDS.REC_TYPE);

        let curPara = JSON.parse(json)
        for (const key in curPara) {
            if (Object.prototype.hasOwnProperty.call(curPara, key)) {
                if (params[key]) {
                    curPara[key] = params[key]
                }
            }
        }
        // 处理query参数的格式
        queryParams.forEach(g => {
            if (Object.keys(params).includes(g.name)) {
                g.valueContent = params[g.name]
            }
            if (Object.keys(curPara).includes(g.name)) {
                curPara[g.name] = g.valueContent;
            }
        })
        // console.log(curPara, "new json")
        json = JSON.stringify(curPara)
        // console.log(queryParams, json, "参数"); // 可以查看还有哪些参数可以修改。
    }

    if (window.SET_CMP_LOADING) {
        // 设置组件加载中状态
        window.SET_CMP_LOADING(window.COMPONENT_IDS.REC_TYPE, true);

    }
    return {
        queryParams,
        formParams,
        json,
        requestHeaders
    }
}

function fit(result) {
  // fit用于成功解析数据源响应后回调
  // result为数据源返回结果
  // 一般为JSON字符串，建议先进行JSON.parse(result)操作)

  // console.log(newRes, "案件类型")
  if (window.SET_CMP_LOADING) {
    const params = (window.GET_GLOBAL_PARAM && window.GET_GLOBAL_PARAM()) || {};
    // console.log('params===>', params)
    //  如果全局参数中startTime、endTime 则还是加载中
    const loading = !params.startTime || !params.endTime

  }
  if (newRes.result) {
    newRes.result.forEach(g => {
      g.archiveRate = (g.archiveRate || 0) + "%"
    })
  }
  return newRes.result;
}

    function getTableCondition(id, type = "") {

        if (param.condition.tableConditionDescriptor && param.condition.tableConditionDescriptor.members) {
            // console.log(param.condition.tableConditionDescriptor.members, "筛选参数")
            const members = param.condition.tableConditionDescriptor.members;

            const curMember = members.find(e => e.field == id);
            if (curMember) {
                switch (curMember.fieldType) {
                    case "string":
                        value = curMember.value
                        break;
                    case "array":
                        value = curMember.values.join(",")
                        if (type === "cascader") {
                            value = curMember.values[curMember.values.length - 1]
                        }
                        break;
                    case "datetime":
                        value = curMember.values[0] ? [`${curMember.values[0]} 00:00:00`, `${curMember.values[1]} 23:59:59`] : ""
                        break;
                }
            }

        }
        return value

function main() {
    const recID = this.$route.query.id || 19
    const self = this
    const PROCESS_ID = "csdn_detail_task_process_1_c92tw0"
    const processCmp = self.$$(PROCESS_ID);
    processCmp.data.props.dataSource.options = [];
    this.$nextTick(() => {
        processCmp.data.props.dataSource.options = [];
        const paramArr = [
            { "name": "id", "valueContent": recID },
        ]
        api.DATA_SOURCE_EXECUTE("d4248521-f50c-483b-bfd7-55f7de097cc7-7d18", "api", paramArr, [], "").then(res => {
            if (res && !res.hasError) {
                const data = JSON.parse(res.result);
                if (data && !data.hasError && data.result) {
                    const list = data.result;
                    const processes = list.processes.map(item => {
                        return {
                            humanName: item.humanName,
                            actDefName: item.actDefName || "",
                            actionName: item.actionName || item.actStateName,
                            actionTime: item.actionTime || item.startTime,
                            bundleRemain: item.bundleRemain,
                            actRemain: item.actRemain,
                            detail: item.detail || '',
                            itemContent: item.processItem.itemContent || "",
                            itemTypeId: item.itemTypeId || "",
                            itemTypeName: item.itemTypeName || "",
                            title: item.title || '',
                            actId: item.actId,
                            unitName: item.unitName
                        }
                    });
                    processCmp.data.props.dataSource.options = processes;
                } else {
                    processCmp.data.props.dataSource.options = [];
                }
            } else {
                processCmp.data.props.dataSource.options = [];
            }
        }).catch(err => {
            processCmp.data.props.dataSource.options = [];
        });
    })
}
