灵珑动作交互&js脚本示例模板
一 灵珑动作交互
灵珑的动作面板提供了内置动作、自定义动作、页面JS，同时提供了前置动和后置动作配置，对于动作面板的基本功能使用此处不做介绍。
具体见《动作面板》
1 元素查找
this 为当前动作上下文；
this.$$("xxxx");     // 从触发当前脚本的组件所在页面的根向下查找id为xxxx的组件元素。
this.$$m("xxxx");  // 从触发当前脚本的组件所在页面的根向下查找id为xxxx的组件模型。
this.$$root();     //查询组件所在部件或页面的最近根元素
this.$$roots();     //查询组件所在部件或页面的所有根元素
this.$$element("xxxx");  //从触发当前脚本的组件层级向上（包括当前）查找id为xxxx的组件元素。
this.$$model("xxxx");  //从触发当前脚本的组件层级向上（包括当前）查找id为xxxx的组件模型。
this.$$element("xxxx", true);  // 从触发当前脚本的组件层级向下（包括当前）查找id为xxxx的组件元素。
this.$$model("xxxx", true);  // 从触发当前脚本的组件层级向下（包括当前）查找id为xxxx的组件模型。
在灵珑脚本中element关键字表示的是灵珑组件元素，一般与页面的可见元素对应，且拥有丰富的内置方法；在灵珑脚本中model关键字表示的是灵珑组件模型，它是组件的数据结构描述，只有属性没有方法。如果组件在属性面板设置隐藏状态之后，页面渲染时并不会渲染该组件，因此无法通过this.$$("组件ID")、this.$$element("组件ID")来找到它，如果需要取组件的属性值，需要用this.$$model("组件ID")。
2 脚本中资源标识（ID）的获取和交互传参说明
灵珑的js脚本中，经常需要引用一些资源的id，例如：部件、页面、数据集、api，把这些id作为参数进行传递。那么，参数如何获取，不同的部件和页面如何传递，请参见文档：《页面交互传参说明【已过时】》

var LoadScript = (function () {
    var instances = {}
    return function (src, callback) {
        if (!instances[src]) {
            instances[src] = new Promise((resolve, reject) => {
                const script = document.createElement('script')
                
                var onload = () => {
                   if (callback) {
                       callback()
                   }
                   resolve()
                }
                
                script.src = src
                script.onload = onload
                script.onerror = reject
                document.head.appendChild(script)
            })
            // 提供一个删除instace的接口
            instances[src].deleteInstance = function () {
                delete instances[src]
            }
        }
        return instances[src]
    }
})()

function main() {
  var element = this;
  var rootPageElement = null;

  while (element) {
      if (element.getPageInfo && element.goHistory) {
          rootPageElement = element;
          break;
      } else {
          element = element.$parent;
      }
  }
  //第一个参数为要打开的页面的pageid，可通过ctrl+shift+c打开资源选择弹窗进行选择
  //isPageHeader为是否开启页头，pageHeaderTitle设置页头的名称,extraParams_设置给页面传参
  rootPageElement.getPageInfo('pageid',"PAGE",{  
    isPageHeader:true,
    pageHeaderTitle:"test",
    extraParams_:{
      params1: "",
      params2: ""
    }
  });
}
通过脚本获取URL参数
该脚本提供了两种方法
function main() {
    // 方式1
  // 获取路由参数
    let query = this.$route.query;
    console.log(query);

  // 方式2
    let location = window.location;
    // 拿到问号后的字符串
    query = location.href.match(/(?<=\?.*).+/g) || [];
    // 根据 & 拆分
    if (query.length > 0) {
        let params = {};
        let paramArr = query[0].split("&");
        paramArr.forEach((p) => {
            // 等号拆分
            let kv = p.split("=");
            params[kv[0]] = decodeURI(kv[1] || "");
        });
        console.log(params);
    }
}

表单中动态控制组件显隐
场景描述：在表单设计器中，用户在操作单选或者下拉单选组件时，当值发送变化，相应的需要控制其他组件的显示和隐藏，如何在自定义动作中编写脚本。

function main() {
    // 找到单选组件元素
    let radioEle = this.$$("radio_39tr41");
  // 获取单选组件内容
    const text = radioEle.getText();
    // 取到被控组件的model
    let textareaModel = this.$$model("textarea_7dimpe");
    if (text === "是") {
      // 设置多行文本组件显示
        // behavior有4种状态分别是 正常：NORMAL,禁用：DISABLED,只读:READONLY,隐藏:HIDDEN
        textareaModel.props.behavior.value = "NORMAL"
    } else if (text === "否") {
      // 设置多行文本隐藏
        textareaModel.props.behavior.value = "HIDDEN"
    }
}

表单中组件自定义校验规则
灵珑表单中的组件校验规则提供一部分常见的内置规则，如必填、身份证号、唯一约束等，内置的校验规则不满足时，可以配置自定义校验规则，通过js脚本，下面示例为车牌号规则校验
function validate(value) {
    /* return支持返回boolean类型或者{ success: boolean, message?: string }对象，
       返回对象格式时success为必填，message为选填，当message有值时错误提示信息将
       改为message设置的提示信息 */
    
    // 新能源车牌
    const xreg = /^[京津沪渝冀豫云辽黑湘皖鲁新苏浙赣鄂桂甘晋蒙陕吉闽贵粤青藏川宁琼使领A-Z]{1}[A-Z]{1}(([0-9]{5}[DF]$)|([DF][A-HJ-NP-Z0-9][0-9]{4}$))/;
    // 旧车牌
    const creg = /^[京津沪渝冀豫云辽黑湘皖鲁新苏浙赣鄂桂甘晋蒙陕吉闽贵粤青藏川宁琼使领A-Z]{1}[A-Z]{1}[A-HJ-NP-Z0-9]{4}[A-HJ-NP-Z0-9挂学警港澳]{1}$/;
    if (value.length === 7) {
        if(value.search(creg)==-1)
        {
          return  { success: false, message: "车牌号格式不正确" } 
        }
        else{
          return  { success: true } 
        }
        
    } else if (value.length === 8) {
        if(value.search(xreg)==-1)
        {
          return  { success: false, message: "车牌号格式不正确" } 
        }
        else{
          return  { success: true } 
        }
    } else {
        return { success: false, message: "车牌号格式不正确" };
    }
}

export function didMount_appListPreview() {
  // 定义支持的文件类型 501:word,503:ppt,502:excel,601:pdf
  var downloadTypes = ["501", "doc", "503", "ppt", "601", "pdf"];
  api.FILE_HANDLE("preview", function (fileRecord, option) {
    var isDirectDownload = fileRecord.url && fileRecord.mediaType && downloadTypes.includes(fileRecord.mediaType);
    if (!isDirectDownload) {
      isDirectDownload = fileRecord.url && fileRecord.type && downloadTypes.includes(fileRecord.type)
    }
    if (isDirectDownload) {
      const url = location.origin + fileRecord.url;
      if (url.indexOf("preview-video") > 0) {
        url = url.replace("/preview-video", "/download");
      }
      window.open(url, "_blank");
      return true;
    }
    return false;
  });
}

列表页面新增按钮打开新增页面的回退事件基于条件判断是否刷新列表

function jumpToPage(self, tableComp) {
    var element = self;
    var rootPageElement = null;
    // 循环向上找页面根元素，就是有页面回退方法的元素
    while (element) {
        if (element.getPageInfo && element.goHistory) {
            rootPageElement = element;
            break;
        } else {
            element = element.$parent;
        }
    }
    if (rootPageElement) {
        // 定义页面路由参数
        const params = {
            extraParams_: { actionType: { action: "send"} }
        };
        // 执行页面跳转，将定义的路由参数传进去
        rootPageElement.getPageInfo("c31d346e-8edd-4b97-89b1-8f8ef76e0b96", "PAGE", params, () => {
            console.log("下级页面回退后动作回调", params);
            // 基于下级页面回退后返回的参数值进行判断，当时提交类型的操作时，执行列表刷新，否则不执行
            if (params && params.extraParams_.actionType.action === "commit"){
                if (tableComp) {
                    tableComp.$$refreshData();
                }
            }          
        });
        // 设置需要传递给下级页面的参数
        rootPageElement.extra['text_input_2s2pde'] = "参数值";
    }
}

树部件开启复选后数据加载时选中指定的节点
在树部件上开启复选之后，希望树加载时选中指定的一批节点，可以在树面板组件的数据变化事件（on-data-change）上增加如下脚本
function main(){
    //获取树数据list
    const list = this.smoothTree(this.renderTreeData);
    const checkedKeys = [];
    const selectedData = [];
    //获取主键列名
    const idColumName = this.getIdColumnName();
    //循环判断哪些节点选中
    list.forEach(x => {
        //先判断节点可被选中
        if (x.checkable) {
            //获取节点的key值
            const nodeKy = x[idColumName] || x.data[idColumName];
            //判断key值是否为指定的要选中的key
            if(nodeKy == '21' || nodeKy == '22')
            {
                checkedKeys.push(nodeKy);
                selectedData.push(x);
            }         
        }
    });
    this.checkedKeys = checkedKeys;
    this.selectedData = selectedData;
}
使用 addEventListener
如果在脚本中直接使用 addEventListener，由于没有对应的 removeEventListener，会存在内存泄漏风险
在任务 450566中，提供 $$addEventListener方法，其 API 与 window.addEventListener 保持一致，调用后在组件销毁时会执行 removeEventListener，避免内存泄漏问题
该方法已经挂载到 BaseElement 上，即任意组件均可使用
示例代码
function main() {
    this.$$addEventListener(
        "message",
        (event) => {
            console.log("receive data", event);
        }
    )
}
