如何配置地图底图的工作台
业务场景说明：
珠海项目需要配置地图为地图且包含交互的工作台。悟空配置情况下字体会压扁并且卡顿，同时交互不好配置
故采用全灵珑配置方式
地图置底配置网格布局:
地图拖入页面，开启网格，配置比例，如5-80:7:7:5-80
左右2个网格项拖入容器
统计类卡片配置
统计类卡片配置一般使用详情部件，配置不同网格容器的背景色、字段、交互等
动态统计数据或列表类配置
如果数量不确定，需采用卡片列表的方式配置，根据不同的数据展示不同的背景色或图片。
容器颜色自定义。卡片数据加载完成时触发js动作(预览时会失效，页面内正常)
export function didMount() {

  // 打印日志：输出当前 this 指向，方便查看作用域
  console.log("this " + this);
  
  // 定义颜色数组，存放3种背景色（淡蓝、淡红、淡绿）
  let collorArr = ['#ebf7ff', '#fff1ee', '#ebf9f7'];
  
  // 获取页面上所有 class 为 u-card-body-wrapper 的元素集合（卡片内容区）
  var contains = document.getElementsByClassName('u-card-body-wrapper');
  
  // 循环遍历获取到的所有卡片元素
  for (index in contains) {
    // 给每一个卡片设置背景色：按顺序使用颜色数组里的颜色
    contains[index].style.backgroundColor = collorArr[index]
  }
}

// 函数功能：初始化左侧第二个ECharts图表（分组柱状图：金湾/斗门/香洲区 三类单位数量）
function initleft2(self) {
  // 获取图表容器DOM元素
  var chartDom = document.getElementById('container_1ba6ni_csshandler_generate');
  // 初始化ECharts图表实例
  var myChart = echarts.init(chartDom);
  // 定义图表配置项变量
  var option;

  // 调用平台数据源API，获取图表数据
  api.DATA_SOURCE_EXECUTE("01a60a37-f0e7-44e7-bc49-50911dc85480", "api", [], [], "").then(res => {
    // 控制台打印接口返回原始数据
    console.log(res);
    // 解析接口返回的JSON字符串，转为JS对象
    var result = JSON.parse(res['result']);

    // 判断：数据存在且无错误
    if (result && !result.hasError) {
      // 获取接口返回的业务数据
      var obj = result.result

      // 初始化变量：金湾区 排放单位数量
      const ys1 = 0;
      // 初始化变量：金湾区 受纳单位数量
      const sn1 = 0;
      // 初始化变量：金湾区 运输企业数量
      const pf1 = 0;

      // 初始化变量：斗门区 排放单位数量
      const ys2 = 0;
      // 初始化变量：斗门区 受纳单位数量
      const sn2 = 0;
      // 初始化变量：斗门区 运输企业数量
      const pf2 = 0;

      // 初始化变量：香洲区 排放单位数量
      const ys3 = 0;
      // 初始化变量：香洲区 受纳单位数量
      const sn3 = 0;
      // 初始化变量：香洲区 运输企业数量
      const pf3 = 0;

      // 循环遍历接口返回的数据对象
      for (const key in obj) {
        // 过滤：只遍历对象自身属性，排除继承属性
        if (obj.hasOwnProperty(key)) {
          // 判断数据类型：排放单位(site)
          if (obj[key].type == 'site') {
            // 判断区域：金湾区 → 赋值排放单位数量
            if (obj[key].name == '金湾区') {
              pf1 = obj[key].value
            }
            // 判断区域：斗门区 → 赋值排放单位数量
            if (obj[key].name == '斗门区') {
              pf2 = obj[key].value
            }
            // 判断区域：香洲区 → 赋值排放单位数量
            if (obj[key].name == '香洲区') {
              pf3 = obj[key].value
            }
          }

          // 判断数据类型：受纳单位(consumptive)
          if (obj[key].type == 'consumptive') {
            // 判断区域：金湾区 → 赋值受纳单位数量
            if (obj[key].name == '金湾区') {
              sn1 = obj[key].value
            }
            // 判断区域：斗门区 → 赋值受纳单位数量
            if (obj[key].name == '斗门区') {
              sn2 = obj[key].value
            }
            // 判断区域：香洲区 → 赋值受纳单位数量
            if (obj[key].name == '香洲区') {
              sn3 = obj[key].value
            }
          }

          // 判断数据类型：运输企业(unit)
          if (obj[key].type == 'unit') {
            // 判断区域：金湾区 → 赋值运输企业数量
            if (obj[key].name == '金湾区') {
              ys1 = obj[key].value
            }
            // 判断区域：斗门区 → 赋值运输企业数量
            if (obj[key].name == '斗门区') {
              ys2 = obj[key].value
            }
            // 判断区域：香洲区 → 赋值运输企业数量
            if (obj[key].name == '香洲区') {
              ys3 = obj[key].value
            }
          }
        }
      }

      // ECharts图表配置项
      option = {
        // 图例配置
        legend: {},
        // 悬浮提示框配置
        tooltip: {},
        // 图表网格边距配置
        grid: {
          left: '3%',      // 左边距
          right: '4%',     // 右边距
          bottom: '3%',    // 下边距
          containLabel: true // 包含标签计算间距
        },
        // 数据集：统一管理图表数据
        dataset: {
          // 维度定义：分类 + 三个指标
          dimensions: ['product', '排放单位', '受纳单位', '运输企业'],
          // 数据源：三个区域 + 对应数量
          source: [
            { product: '金湾区', 排放单位: pf1, 受纳单位: sn1, 运输企业: ys1 },
            { product: '斗门区', 排放单位: pf2, 受纳单位: sn2, 运输企业: ys2 },
            { product: '香洲区', 排放单位: pf3, 受纳单位: sn3, 运输企业: ys3 },
          ]
        },
        // X轴：类目轴（文字）
        xAxis: { type: 'category' },
        // Y轴：数值轴（自动计算）
        yAxis: {},
        // 系列：3个柱状图
        series: [{ type: 'bar' }, { type: 'bar' }, { type: 'bar' }]
      };

      // 将配置项应用到图表，渲染显示
      option && myChart.setOption(option);
    } else {
      // 数据异常：页面弹出错误提示
      self.$message.error(result.message);
    }
  }).catch(err => {
    // 接口请求失败：控制台打印错误
    console.error(err);
  });
}

  var option;

  // 数组：存储上月数据
  const tArr = [];
  // 数组：存储本月数据
  const pArr = [];

      // 业务数据
      var data = result.result

      // 循环：提取上月数据

        tArr.push(data.pMonth[i])

function showSite(self) {
  // 请求点位接口
  api.DATA_SOURCE_EXECUTE("57dd2a8c-8793-4ce9-b1d9-cdad946b13d9", "api", [], [], "").then(res => {
    console.log(res);
    var result = JSON.parse(res['result']);

    if (result && !result.hasError) {
      // 数据格式化：处理经纬度、名称、区域
      var list = result.result.map(row => {
        return {
          id: row.id,               // 点位ID
          y: row.y_coord,           // 纬度
          x: row.x_coord,           // 经度
          '名称': row.name,         // 点位名称
          '所属区域': row.region_name, // 区域
