指标卡初次加载选中第一个卡片，数据刷新后保持之前选中的卡片不变
在卡片列表的卡片容器组件上添加onListChange事件，执行如下脚本：
如果之前没有选中则在数据变化时选中第一个卡片，如果之前有选中，列表刷新之后会清除选中状态，因此需要在数据变化时再执行一次选中
function main(){
    var self = this.$$("card_container_abcdef");

    var index = 0;
    // 获取之前已选中的卡片
    var selection = self.$$getSelection();

    // 取已选中卡片的key
    if (selection.selectedRowKeys.length > 0 )
        index = selection.selectedRowKeys[0];
    // 基于index执行卡片选中，之前没有选中则选第一个，之前有选中则仍选中之前的
    if (self && self.dataList && self.dataList.length && self.onCheckedChange) {
        var item = self.dataList[index];
        self.onCheckedChange("radio", item, true);
    }
}
（3）新增数据交互：
点击数据采集按钮进入新增页面时，需要获取树中当前选中的街道和指标卡中当前选中的主体类型作为参数进行传参；注意：只有树选中的节点为街道级节点时才传递参数。调整页面的参数基于自定义参数
function main() {
    var pageTableList = this.$$("page_widget_list_abcde");
    // 获取列表渲染页实例
    var listRenderView = pageTableList.$$getRenderElement();
    // 获取当前显示的组件，表格或者卡片
    var currentListEle = listRenderView.$$getListElement();
    // 获取当前选中
    var cardSelect = currentListEle.$$getSelection();
    var type_id = "";
    if (cardSelect.selectedRows.length > 0)
        type_id = cardSelect.selectedRows[0].litigant_big_type_code.value;

    var pageTreeEle = this.$$("page_tree_abcdef");
    var treeRenderEle = pageTreeEle.$$getRenderElement();
    var treePanelEle = treeRenderEle.$$getTreeElement();
    var street_id = "";
    if (treePanelEle.$$getSelection().selectedKeys.length != 0) {
        if (treePanelEle.$$getSelection().selectedRows[0].data.columns.type == 3) {
            street_id = treePanelEle.$$getSelection().selectedKeys[0];
        }
    }
    if (street_id != "")
        return {
            'street_id': street_id,
            'type_id': type_id
        };
    else
        return { 'type_id': type_id};
}

function main(){
    // 查找页面参数
    var rootEleParams = this.$$roots()[0].$$getRenderElement().extraParams;
    if (rootEleParams) 
    {
        var type_id = rootEleParams.type_id;
        if (type_id && type_id === "1") {
            //设置指定索引的标签页是否可见，序号从1开始
            //设置指定索引的标签页选中
            this.params.currentTab = 3;
            this.data.children[1].props.visible = false;
            this.data.children[2].props.visible = false;
        }
        else if (type_id && type_id === "2")
        {
            //设置指定索引的标签页是否可见，序号从1开始
            this.data.children[2].props.visible = false;
            this.data.children[3].props.visible = false;
            //设置指定索引的标签页选中
            this.params.currentTab = 1;
        }
        else if (type_id && type_id === "3") {
            //设置指定索引的标签页是否可见，序号从1开始
            this.data.children[1].props.visible = false;
            this.data.children[3].props.visible = false;
            //设置指定索引的标签页选中
            this.params.currentTab = 2;
        }
    }
        
}

export function didMount() {
  console.log("「页面 JS」：当前页面地址 " + location.href);
  var rootEleParams = this.$$roots()[0].$$getRenderElement().extraParams;
  let form_part1 = this.$$m("form_part_123456");
  let form_part2 = this.$$m("form_part_abcdef");
  let form_part3 = this.$$m("form_part_654321");
  if (rootEleParams) {
    var type_id = rootEleParams.type_id;
    if (type_id && type_id === "1") {
      form_part1.props.hide = false;
    }
    else if (type_id && type_id === "2") {
      form_part2.props.hide = false;
    }
    else if (type_id && type_id === "3") {
      form_part3.props.hide = false;
    }
  }
}
