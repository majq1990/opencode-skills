## 实时数据对接脚本

### 使用说明

- 使用空接口 `dex-api/free/req-resp` 触发脚本。
- 这是合并脚本，仍处理 9 类实时业务数据。
- `page`、`count` 只用于 `BridgeAcceSensorMonitor` 分页。
- 当 `page = 0` 时，脚本会同步其他 8 类小数据业务；当 `page != 0` 时，其他 8 类会直接跳过，避免分页过程中重复推送或源端高页码报错。
- 后置脚本返回 `totalSize` 和 `bridgeAcceSensorMonitorTotalSize`，值都取 `BridgeAcceSensorMonitor` 源端返回的 `data.total`，用于定时任务计算分页。
- 脚本中涉及的 IP、端口、`clientId`、`clientSecret` 部署时需要按现场环境替换。

调用 body 示例：

```json
{
  "startTime": "2026-05-21 10:00:00",
  "endTime": "2026-05-21 10:10:00",
  "page": "0",
  "count": "1000"
}
```

### 前置脚本

```groovy
import org.springframework.http.HttpHeaders
import com.egova.api.util.http.HttpUtils
import com.egova.json.utils.JsonUtils
import java.text.SimpleDateFormat

// ========== 现场配置：部署时需要替换地址、端口和认证参数 ==========
var sourceBaseUrl = 'http://59.208.239.152:38847'
var iotBaseUrl = 'http://172.16.142.37:6888/iot'
var clientId = '78f92ae2-b7e5-4521-884c-312a1becfb3c'
var clientSecret = '5249f4dc-b400-4899-8746-9ced6387fa75'

var tokenUrl = sourceBaseUrl + '/gsc-ishare/v2/itopdm/upms/oauth/token'
var sourceGetUrl = sourceBaseUrl + '/gsc-ishare/v2/bztp/get'
var iotRealtimeUrl = iotBaseUrl + '/free/exchange/equip/list'
var bridgeAcceEntity = 'BridgeAcceSensorMonitor'

// 本次实时数据对接范围
var realtimeConfigs = [
    ['biSigns': 'bztp-cl-rtm-GasMonitor', 'entity': 'GasMonitor', 'fields': [
        ['source': 'ch4_value', 'target': 'ch4_value'],
        ['source': 'ch4_state', 'target': 'ch4_state', 'alarmWhenOne': true]
    ]],
    ['biSigns': 'bztp-cl-rtm-WaterPipeFlowmeterMonitor', 'entity': 'WaterPipeFlowmeterMonitor', 'fields': [
        ['source': 'instant', 'target': 'instant']
    ]],
    ['biSigns': 'bztp-cl-rtm-WaterPipePressureMonitor', 'entity': 'WaterPipePressureMonitor', 'fields': [
        ['source': 'press', 'target': 'press']
    ]],
    ['biSigns': 'bztp-cl-rtm-DrainPipeLevelMonitor', 'entity': 'DrainPipeLevelMonitor', 'fields': [
        ['source': 'water_level', 'target': 'water_level']
    ]],
    ['biSigns': 'bztp-cl-rtm-DrainPipeLevelSample5M', 'entity': 'DrainPipeLevelSample5M', 'fields': [
        ['source': 'water_level', 'target': 'water_level']
    ]],
    ['biSigns': 'bztp-cl-rtm-BridgeAcceSensorMonitor', 'entity': 'BridgeAcceSensorMonitor', 'fields': [
        ['source': 'acceleration', 'target': 'acceleration']
    ]],
    ['biSigns': 'bztp-cl-rtm-BridgeHygrometerMonitor', 'entity': 'BridgeHygrometerMonitor', 'fields': [
        ['source': 'humidity', 'target': 'humidity']
    ]],
    ['biSigns': 'bztp-cl-rtm-BridgeTemperSensorMonitor', 'entity': 'BridgeTemperSensorMonitor', 'fields': [
        ['source': 'temperature', 'target': 'temperature']
    ]],
    ['biSigns': 'bztp-cl-rtm-BridgeStrainSensorMonitor', 'entity': 'BridgeStrainSensorMonitor', 'fields': [
        ['source': 'strain', 'target': 'strain']
    ]]
]

// ========== 工具方法区 ==========
var hasText = { value ->
    return value != null && String.valueOf(value).trim() != ''
}

var toText = { value ->
    return value == null ? null : String.valueOf(value).trim()
}

var toDouble = { value ->
    if (!hasText(value)) {
        return null
    }
    try {
        return Double.parseDouble(String.valueOf(value).trim())
    } catch (Exception ignored) {
        return null
    }
}

var toLong = { value ->
    if (value == null) {
        return null
    }
    if (value instanceof Number) {
        return ((Number) value).longValue()
    }
    if (!hasText(value)) {
        return null
    }
    try {
        return Long.parseLong(String.valueOf(value).trim())
    } catch (Exception ignored) {
        return null
    }
}

var formatTime = { millis ->
    var sdf = new SimpleDateFormat('yyyy-MM-dd HH:mm:ss')
    return sdf.format(new Date(millis))
}

var toMillis = { value ->
    if (value == null) {
        return null
    }
    if (value instanceof Number) {
        return ((Number) value).longValue()
    }

    var timeStr = String.valueOf(value).trim()
    if (timeStr == '') {
        return null
    }

    try {
        return Long.parseLong(timeStr)
    } catch (Exception ignored) {
    }

    var patterns = ['yyyy-MM-dd HH:mm:ss', 'yyyy-MM-ddHH:mm:ss', 'yyyy/MM/dd HH:mm:ss']
    for (pattern in patterns) {
        try {
            var sdf = new SimpleDateFormat(pattern)
            return sdf.parse(timeStr.take(19)).getTime()
        } catch (Exception ignored) {
        }
    }
    return null
}

var removeEmpty = { sourceMap ->
    return sourceMap.findAll { entry ->
        entry.value != null && !(entry.value instanceof String && entry.value.trim() == '')
    }
}

var getSourceToken = {
    var params = [
        'client_id'    : clientId,
        'client_secret': clientSecret,
        'scope'        : 'all',
        'grant_type'   : 'client_credentials'
    ]
    var tokenRes = HttpUtils.postJson(
        tokenUrl,
        params,
        Map.class,
        { HttpHeaders headers ->
            headers.set('Content-Type', 'application/json')
        }
    ).getBody()
    return tokenRes.data.access_token
}

var postSourceQuery = { accessToken, biSigns, bodyMap ->
    return HttpUtils.postJson(
        sourceGetUrl,
        bodyMap,
        Map.class,
        { HttpHeaders headers ->
            headers.set('Content-Type', 'application/json')
            headers.set('Authorization', 'Bearer ' + accessToken)
            headers.set('biSigns', biSigns)
        }
    ).getBody()
}

var postIotRealtime = { pushBody ->
    return HttpUtils.postJson(
        iotRealtimeUrl,
        pushBody,
        Map.class,
        { HttpHeaders headers ->
            headers.set('Content-Type', 'application/json')
        }
    ).getBody()
}

// ========== 主逻辑区 ==========
variables['pushFlag'] = false
variables['pushMsg'] = ''
variables['pushResult'] = ''
variables['totalSize'] = 0L
variables['bridgeAcceSensorMonitorTotalSize'] = 0L
variables['pushSize'] = 0
variables['page'] = '0'
variables['count'] = '1000'
variables['startTime'] = ''
variables['endTime'] = ''
variables['querySummary'] = ''
variables['errorSummary'] = ''

try {
    var nowMillis = new Date().time
    var startTime = formatTime(nowMillis - 10 * 60 * 1000)
    var endTime = formatTime(nowMillis)
    var page = '0'
    var count = '1000'

    // 读取定时任务传入的时间范围和分页参数
    var bodyStr = request.getBody().getString()
    if (hasText(bodyStr)) {
        var input = JsonUtils.deserialize(bodyStr, Map.class)
        if (input instanceof Map) {
            if (hasText(input.get('startTime'))) {
                startTime = String.valueOf(input.get('startTime'))
            }
            if (hasText(input.get('endTime'))) {
                endTime = String.valueOf(input.get('endTime'))
            }
            if (hasText(input.get('page'))) {
                page = String.valueOf(input.get('page'))
            }
            if (hasText(input.get('count'))) {
                count = String.valueOf(input.get('count'))
            }
        }
    }

    variables['page'] = page
    variables['count'] = count
    variables['startTime'] = startTime
    variables['endTime'] = endTime

    var accessToken = getSourceToken()
    if (!hasText(accessToken)) {
        variables['pushMsg'] = '源端token为空，未推送'
        return 'api_stop'
    }

    var pushSize = 0
    var pushResults = []
    var querySummary = []
    var errorSummary = []

    for (config in realtimeConfigs) {
        var entityName = config.entity

        // BridgeAcceSensorMonitor 按传入 page/count 分页；其他小数据业务只在第一页处理一次
        if (entityName != bridgeAcceEntity && page != '0') {
            querySummary.add(entityName + ':skip')
            continue
        }

        try {
            var queryBody = [
                '[]'    : [
                    (entityName): [
                        'time>=': startTime,
                        'time<=': endTime,
                        '@order': 'time-'
                    ],
                    'page'    : page,
                    'count'   : count,
                    'query'   : '2'
                ],
                'total@': '/[]/total'
            ]

            // 非 BridgeAcceSensorMonitor 的小数据业务固定查第一页，避免跟随桥梁加速度分页导致高页码异常
            if (entityName != bridgeAcceEntity) {
                queryBody['[]']['page'] = '0'
                queryBody['[]']['count'] = '1000'
            }

            var sourceRes = postSourceQuery(accessToken, config.biSigns, queryBody)
            var dataNode = sourceRes.data instanceof Map ? sourceRes.data : [:]
            var sourceList = dataNode['[]'] instanceof List ? dataNode['[]'] : []
            var sourceTotal = toLong(dataNode.total)
            if (sourceTotal == null) {
                sourceTotal = 0L
            }

            if (entityName == bridgeAcceEntity) {
                variables['totalSize'] = sourceTotal
                variables['bridgeAcceSensorMonitorTotalSize'] = sourceTotal
            }

            var pushBody = []
            for (itemWrapper in sourceList) {
                if (!(itemWrapper instanceof Map)) {
                    continue
                }

                var item = itemWrapper.get(entityName)
                if (!(item instanceof Map)) {
                    continue
                }

                var equipCode = toText(item.equipCode)
                if (!hasText(equipCode)) {
                    equipCode = toText(item.pointCode)
                }
                if (!hasText(equipCode)) {
                    continue
                }

                var pushTime = toMillis(item.time)
                if (pushTime == null) {
                    pushTime = toMillis(item.uptime)
                }
                if (pushTime == null) {
                    pushTime = nowMillis
                }

                var dataList = []
                for (field in config.fields) {
                    var sourceField = field.source
                    if (!item.containsKey(sourceField) || item[sourceField] == null || String.valueOf(item[sourceField]).trim() == '') {
                        continue
                    }

                    var dataItem = [
                        'fieldCode': field.target,
                        'value'    : toDouble(item[sourceField])
                    ]

                    if (field.alarmWhenOne == true) {
                        dataItem['alarm'] = String.valueOf(item[sourceField]).trim() == '1' ? 1 : 0
                    }

                    dataList.add(removeEmpty(dataItem))
                }

                if (dataList.isEmpty()) {
                    continue
                }

                // 转换为物联网 1.0 实时数据结构
                pushBody.add([
                    'equipCode': equipCode,
                    'time'     : pushTime,
                    'data'     : dataList
                ])
            }

            if (!pushBody.isEmpty()) {
                var iotRes = postIotRealtime(pushBody)
                pushSize = pushSize + pushBody.size()
                pushResults.add([
                    'entity'    : entityName,
                    'pushSize'  : pushBody.size(),
                    'pushResult': iotRes
                ])
            }

            querySummary.add(entityName + ':' + pushBody.size() + '/' + sourceTotal)
        } catch (Exception itemError) {
            // 单类业务失败只记录错误，不影响其他业务继续执行
            errorSummary.add(entityName + ':' + itemError.message)
            querySummary.add(entityName + ':error')
        }
    }

    variables['pushSize'] = pushSize
    variables['querySummary'] = querySummary.join('；')
    variables['errorSummary'] = errorSummary.join('；')
    variables['pushResult'] = JsonUtils.serialize(pushResults)

    if (pushSize > 0) {
        variables['pushFlag'] = true
        variables['pushMsg'] = '实时数据处理完成，推送数量：' + pushSize + '，BridgeAcceSensorMonitor总数：' + variables['bridgeAcceSensorMonitorTotalSize']
    } else {
        variables['pushMsg'] = '未找到可推送的实时数据，BridgeAcceSensorMonitor总数：' + variables['bridgeAcceSensorMonitorTotalSize']
    }

    return 'api_stop'
} catch (Exception e) {
    variables['pushMsg'] = '实时数据脚本执行异常：' + e.message
    return 'api_stop'
}
```

### 后置脚本

```groovy
import com.egova.json.utils.JsonUtils

// 返回分页和推送结果，定时任务按 totalSize 计算 BridgeAcceSensorMonitor 页数
var result = [
    'pushFlag'                         : variables['pushFlag'],
    'pushMsg'                          : variables['pushMsg'],
    'totalSize'                        : variables['totalSize'],
    'bridgeAcceSensorMonitorTotalSize' : variables['bridgeAcceSensorMonitorTotalSize'],
    'pushSize'                         : variables['pushSize'],
    'page'                             : variables['page'],
    'count'                            : variables['count'],
    'startTime'                        : variables['startTime'],
    'endTime'                          : variables['endTime'],
    'querySummary'                     : variables['querySummary'],
    'errorSummary'                     : variables['errorSummary'],
    'pushResult'                       : variables['pushResult']
]

return JsonUtils.serialize(result)
```
