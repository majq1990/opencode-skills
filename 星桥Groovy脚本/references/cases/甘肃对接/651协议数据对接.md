编写星桥脚本对接设备实时数据

数据格式：
{"data":{
  "rawHex": "7E7E014132260071123432002F...",
  "frameHead": "7E7E",
  "centerAddress": "01",
  "stationAddress": "4132260071",
  "password": "1234",
  "functionCode": "32",
  "directionAndLength": "002F",
  "bodyLength": 47,
  "bodyStartFlag": "02",
  "serialNumber": 5,
  "reportTime": "2026-04-27 17:00:29",
  "reportTimeMillis": 1777280429000,
  "bodyStationAddress": "4132260071",
  "stationTypeCode": "48",
  "stationTypeName": "河道",
  "observeTime": "2026-04-27 17:00:00",
  "observeTimeMillis": 1777280400000,
  "bodyEndFlag": "03",
  "crcHex": "55AE",
  "crcValid": true,
  "packetType": "LONG",
  "measurements": []
}}

监测数据字段示例：

```json
{
  "elementCode": "3923",
  "formatCode": "23",
  "rawValueHex": "00000250",
  "dataLength": 4,
  "decimalPlaces": 3,
  "value": 0.25
}
```
字段说明：

- `packetType=LONG`：长报文，包含监测数据。
- `packetType=SHORT`：短报文或心跳报文，正常回复确认报文，`measurements` 为空。
- `elementCode`：协议原始监测要素码，例如 `3923`、`3E23`、`3812`、`7A08`、`FFA011`。
- `value`：按格式码解析并缩放后的数值。
- `rawValueHex`：原始监测值 HEX，便于外部系统复核。

其中 stationAddress 是equipCode

监测项编码需要映射，
例如：3923 是 water_level
 3913 也是 water_level
 FFA112 是 battery_voltage
 3812 是 battery_voltage
保留映射能力，方便后续扩展。

## 星桥脚本配置说明

- 对接场景：设备实时数据。
- 星桥接口：使用空接口 `dex-api/free/req-resp` 触发前置脚本和后置脚本。
- 入参方式：将 651 协议解析后的 JSON 作为空接口 body 传入。
- 推送目标：前置脚本主动调用 IoT 实时数据接口 `/api/exchange/equip/list`。
- 短报文处理：`packetType=SHORT` 或 `measurements` 为空时不推送实时数据，直接结束空接口调用。
- 需要按现场替换脚本中的 `tokenUrl`、`iotBaseUrl`、`clientId`、`clientSecret`。

## 前置脚本

```groovy
import org.springframework.http.HttpHeaders
import com.egova.api.util.http.HttpUtils

// ========== 现场配置：部署时需要替换 ==========
var tokenUrl = 'http://10.161.133.68:10050/usercenter-api/oauth/extras/token'
var iotBaseUrl = 'http://173.16.0.17:8084/iot-api'
var clientId = 'usercenter'
var clientSecret = '54ddc335f9'

// 651 要素码到 IoT 监测项编码的映射，后续新增字段只需要扩展这里
var fieldCodeMap = [
    '3923'  : 'water_level',
    '3913'  : 'water_level',
    'FFA112': 'battery_voltage',
    '3812'  : 'battery_voltage'
]

// 读取空接口 body，body 格式支持 {"data": {...}} 或直接传 {...}
var bodyStr = request.getBody().getString()
var bodyMap = com.egova.json.utils.JsonUtils.deserialize(bodyStr, Map.class)
var equipData = bodyMap?.data instanceof Map ? bodyMap.data : bodyMap

variables['pushFlag'] = false
variables['pushMsg'] = ''
variables['pushBody'] = ''
variables['pushResult'] = ''

if (equipData == null) {
    variables['pushMsg'] = 'body 为空，未推送'
    return 'api_stop'
}

var packetType = equipData.packetType == null ? '' : equipData.packetType.toString()
var measurements = equipData.measurements instanceof List ? equipData.measurements : []
if (packetType == 'SHORT' || measurements.isEmpty()) {
    variables['pushMsg'] = '短报文或无监测数据，未推送'
    return 'api_stop'
}

var equipCode = equipData.stationAddress == null ? null : equipData.stationAddress.toString()
var dataList = []

measurements.each { item ->
    var elementCode = item.elementCode == null ? '' : item.elementCode.toString().toUpperCase()
    var targetFieldCode = fieldCodeMap[elementCode]
    if (targetFieldCode != null) {
        var detailMap = [
            'elementCode' : elementCode,
            'rawValueHex' : item.rawValueHex,
            'formatCode'  : item.formatCode,
            'dataLength'  : item.dataLength
        ]

        dataList.add([
            'fieldCode': targetFieldCode,
            'value'    : item.value
        ])
    }
}

if (equipCode == null || dataList.isEmpty()) {
    variables['pushMsg'] = '设备编码为空或无可映射监测项，未推送'
    return 'api_stop'
}

var pushTime = equipData.observeTimeMillis
if (pushTime == null) {
    pushTime = equipData.reportTimeMillis
}
if (pushTime == null) {
    pushTime = System.currentTimeMillis()
}

var pushBody = [[
    'equipCode': equipCode,
    'time'     : pushTime,
    'data'     : dataList
]]

// 获取 IoT token
var tokenParams = [
    'url'         : tokenUrl,
    'clientId'    : clientId,
    'clientSecret': clientSecret
]
var iottoken = tokenStore.load('default', tokenParams).value

// 主动推送到设备实时数据接口
var iotRes = HttpUtils.postJson(
    iotBaseUrl + '/api/exchange/equip/list',
    pushBody,
    Map.class,
    { HttpHeaders headers ->
        headers.set('Content-Type', 'application/json')
        headers.set('Authorization', 'bearer ' + iottoken)
    }
).getBody()

variables['pushFlag'] = true
variables['pushMsg'] = '实时数据已推送'
variables['pushResult'] = com.egova.json.utils.JsonUtils.serialize(iotRes)

return 'api_stop'
```

## 后置脚本

```groovy
// 前置脚本使用 api_stop 时通常不会继续请求空接口；如平台仍执行后置脚本，则返回推送结果
var result = [
    'pushFlag'  : variables['pushFlag'],
    'pushMsg'   : variables['pushMsg'],
    'pushResult': variables['pushResult']
]

return com.egova.json.utils.JsonUtils.serialize(result)
```

## 推送结果示例

输入中 `stationAddress` 会作为 `equipCode`，`observeTimeMillis` 优先作为实时数据时间。`measurements` 中已配置映射的要素会被转换为 IoT 监测项：

```json
  }
]
```
