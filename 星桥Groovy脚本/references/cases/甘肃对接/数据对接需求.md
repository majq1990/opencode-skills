```前置脚本
import org.springframework.http.HttpHeaders
import com.egova.api.util.http.HttpUtils
import com.egova.json.utils.JsonUtils
import java.text.SimpleDateFormat

// 获取 iot2.0 的 token（实际使用请替换 IP 和端口）
var type = 'default'
var params = [
    'url': 'http://10.0.0.10:8080/usercenter-api/oauth/extras/token',
    'clientId': 'usercenter',
    'clientSecret': '54ddc335f9'
]
var iottoken = tokenStore.load(type, params)
variables['iottoken'] = iottoken.value

// 读取请求体
var bodystr = request.getBody().getString()
var bodyMap = JsonUtils.deserialize(bodystr, Map.class)

def equipCode = null
def time = null
def dataList = []

// 格式A：dataReport（含 payload）
if (bodyMap?.payload instanceof Map) {
    def payload = bodyMap.payload
    equipCode = bodyMap.deviceId
    time = bodyMap.timestamp

    // 若主时间为空，则尝试解析 payload.time（yyyyMMddHHmmss）
    if (time == null && payload.time != null && payload.time.toString() != '') {
        def sdf = new SimpleDateFormat("yyyyMMddHHmmss")
        time = sdf.parse(payload.time.toString()).getTime()
    }

    // 排除无需上报字段
    def excludeFields = ['time', 'sensor_fre', 'net_fre', 'alarm_fre', 'Switch_data', 'software_version']
    // 字段映射：IMEI/ICCID 走 detail
    def fieldMapping = [
        'liquid_value1': [newFieldCode: 'liquid_value1', hasDetail: false],
        'IMEI'         : [newFieldCode: 'imei',          hasDetail: true],
        'ICCID'        : [newFieldCode: 'iccid',         hasDetail: true]
    ]

    payload.each { key, value ->
        if (excludeFields.contains(key)) return
        if (value == null || value == '') return

        def mapping = fieldMapping.get(key)
        def fieldCode = mapping ? mapping.newFieldCode : key
        def hasDetail = mapping ? mapping.hasDetail : false

        def item = [:]
        item.fieldCode = fieldCode
        if (hasDetail) {
            item.detail = value.toString()
        } else {
            item.value = value
        }
        dataList << item
    }
}
// 格式B：新增格式（device_code + 4 个监测项）
else {
    equipCode = bodyMap.device_code
    time = bodyMap.timestamp

    ['characteristic_value', 'center_frequency', 'vbat', 'gain'].each { key ->
        def value = bodyMap[key]
        if (value == null || value == '') return
        dataList << [
            fieldCode: key,
            value    : value
        ]
    }
}

// 关键字段缺失时停止
if (equipCode == null || equipCode.toString() == '' || dataList.isEmpty()) {
    variables['iotRes'] = 'equipCode为空或监测项为空，停止推送'
    return 'api_stop'
}

// 无时间字段时，使用当前时间
if (time == null) {
    time = System.currentTimeMillis()
}

def exchangeData = [[
    data     : dataList,
    equipCode: equipCode.toString(),
    time     : time
]]

variables['iotdata'] = exchangeData

// 推送到物联网实时接口（实际使用请替换 IP 和端口）
var iotUrl = 'http://10.0.0.10:8080/iot-api'
var iotRes = HttpUtils.postJson(
    iotUrl + '/api/exchange/equip/list',
    exchangeData,
    String.class,
    { HttpHeaders headers ->
        headers.set('Content-Type', 'application/json')
        headers.set('Authorization', 'bearer ' + variables['iottoken'])
    }
).getBody()

variables['iotRes'] = iotRes
```
