完整脚本如下：
export function init() {
  let that = this;
  const refreshData = () => {

    const errCallBack = () => {
      that.$$m('csdn_page_indicator_9otgae').props.dataSource.options = [];
      that.$$m('csdn_page_indicator_9otgae').props.loadingConfig.loading = false;
    }
    const start = that.$$m('csdn_form_custom_date_gz8j4b').props.currentTime.start.format("YYYY-MM-DD");
    const end = that.$$m('csdn_form_custom_date_gz8j4b').props.currentTime.end.format("YYYY-MM-DD");
    const type = that.$$m('csdn_form_custom_date_gz8j4b').props.defaultValue.value;

    const params = [
      { name: "start", valueContent: start},
      { name: "end", valueContent: end },
      { name: "type", valueContent: type }
    ]

    that.$$m('csdn_page_indicator_9otgae').props.loadingConfig.loading = true;

    api.DATA_SOURCE_EXECUTE("ds_api:d2dabd8d-76fa-45d0-8825-64f5457bdd3c", "api", params, [], "").then(res => {
      if (res && !res.hasError && res.result) {
        const jsonRes = JSON.parse(res.result);
        if (jsonRes && !jsonRes.hasError && jsonRes.result) {
          const data = jsonRes.result;
          that.$$m('csdn_page_indicator_9otgae').props.dataSource.options = data;
          that.$$m('csdn_page_indicator_9otgae').props.loadingConfig.loading = false;
        } else {
          errCallBack();
        }
      } else {
        errCallBack();
      }
    }).catch(err => {
      console.error(err);
      errCallBack();
    });
  }

  refreshData();

  window.REFRESHDATA = refreshData;
}
此处init就是一开始didMount绑定的事件
