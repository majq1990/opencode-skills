1）需要使用自定义渲染函数
function render(h, value, rows) {
    const isVue2 = (this.Vue || this.vue).version.startsWith("2");
    const map = {
        "1": { text: "工作部署", icon: "icon-component-type-graph", v3Icon: "magic-linglong-edit-sp-v2" },
        "2": { text: "材料报送", icon: "icon-upload1", v3Icon: "magic-linglong-feiji" }
    }
    const data = map[value] || map["1"];
    const text = data.text; // 文本信息
    const style = { // 图标样式
        marginRight: "4px"
    }
    if (isVue2) {
        const icon = data.icon;
        return h("div", [h("i", { class: ["iconfont-form-design", icon], style }), h("span", text)])
    }

    const { appContext } = vue.getCurrentInstance();
    const iconfont = appContext.components["u-iconfont-design-form"]
    const icon = data.v3Icon;
    // <u-iconfont-design-form class="icont-form" type="magic-linglong-component-type-form" />
    return h("div", [h(iconfont, { type: icon, class: "icon-form", style }), h("span", text)])
}
